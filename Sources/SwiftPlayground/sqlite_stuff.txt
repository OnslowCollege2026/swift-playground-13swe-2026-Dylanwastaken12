BEGIN TRANSACTION;

PRAGMA FOREIGN_KEYS = OFF;

CREATE TABLE IF NOT EXISTS "Customer" (`CustomerID` INTEGER PRIMARY KEY, `Given Name` TEXT NOT NULL CHECK(LENGTH("Given Name") > 0), `Family Name` TEXT NOT NULL CHECK(LENGTH("Family Name") > 0), `Home Phone` TEXT NOT NULL CHECK(LENGTH("Home Phone") > 0), `Work Phone` TEXT, `Mobile Phone` TEXT, `Address` TEXT NOT NULL CHECK(LENGTH("Address") > 0), `Region` TEXT NOT NULL CHECK(LENGTH("Region") > 0), `City` TEXT NOT NULL, `Country` TEXT NOT NULL CHECK(LENGTH("Country") > 0), `Email` TEXT NOT NULL CHECK(LENGTH("Email") > 0)) STRICT;

CREATE TABLE `Location` (`AirportCode` INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL CHECK(LENGTH("AirportCode") > 0), `Airport Name` TEXT NOT NULL CHECK(LENGTH("Airport Name") > 0), `Airport Adress` TEXT NOT NULL CHECK(LENGTH("Airport Adress") > 0)) STRICT;

CREATE TABLE `Plane` (`PlaneID` INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL CHECK(LENGTH("PlaneID") > 0), `Model` TEXT NOT NULL CHECK(LENGTH("Model") > 0), `Min Runway Length` INTEGER NOT NULL CHECK(LENGTH("Min Runway Length") > 0), `Max Weight` INTEGER NOT NULL CHECK(LENGTH("Max Weight") > 0), `Capacity` INTEGER NOT NULL CHECK(LENGTH("Capacity") > 0)) STRICT;

CREATE TABLE IF NOT EXISTS "Runway" (`RunwayID` INTEGER PRIMARY KEY NOT NULL CHECK(LENGTH("RunwayID") > 0), `Length` INTEGER NOT NULL CHECK(LENGTH("Length") > 0), `AirportCode` INTEGER NOT NULL REFERENCES `Location`(`AirportCode`), `Type` TEXT NOT NULL CHECK(LENGTH("Type") > 0)) STRICT;

CREATE TABLE `Seat` (`SeatID` INTEGER PRIMARY KEY AUTOINCREMENT UNIQUE NOT NULL CHECK(LENGTH("SeatID") > 0), `PlaneID` INTEGER NOT NULL REFERENCES `Plane`(`PlaneID`), `Seat Class` TEXT NOT NULL CHECK(LENGTH("Seat Class") > 0), `Emergency` INTEGER NOT NULL) STRICT;

CREATE TABLE `Flight` (`FlightID` INTEGER PRIMARY KEY AUTOINCREMENT UNIQUE NOT NULL CHECK(LENGTH("FlightID") > 0), `Departure Time` TEXT NOT NULL CHECK(LENGTH("Departure Time") > 0) DEFAULT (00/00/0000), `PlaneID` INTEGER NOT NULL REFERENCES `Plane`(`PlaneID`), `Departure Location` TEXT NOT NULL REFERENCES `Location`(`AirportCode`), `Destination Location` TEXT NOT NULL REFERENCES `Location`(`AirportCode`)) STRICT;

CREATE TABLE `Staff` (`StaffID` INTEGER PRIMARY KEY AUTOINCREMENT UNIQUE NOT NULL CHECK(LENGTH("StaffID") > 0), `Given Name` TEXT NOT NULL CHECK(LENGTH("Given Name") > 0), `Family Name` TEXT NOT NULL CHECK(LENGTH("Family Name") > 0), `Job Title` TEXT NOT NULL CHECK(LENGTH("Job Title") > 0), `Work Email` TEXT NOT NULL CHECK(LENGTH("Work Email") > 0), `Work Phone` TEXT NOT NULL CHECK(LENGTH("Work Phone") > 0), `Mobile Phone` TEXT, `AirportCode` TEXT NOT NULL REFERENCES `Location`(`AirportCode`)) STRICT;

CREATE TABLE `Booking` (`BookingID` INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL CHECK(LENGTH("BookingID") > 0), `CustomerID` INTEGER NOT NULL REFERENCES `Customer`(`CustomerID`), `StaffID` INTEGER NOT NULL REFERENCES `Staff`(`StaffID`), `Passenger Count` INTEGER NOT NULL CHECK(LENGTH("Passenger Count") > 0), `FlightID` INTEGER NOT NULL REFERENCES `Flight`(`FlightID`)) STRICT;

CREATE TABLE `Passenger` (`FlightID` INTEGER NOT NULL REFERENCES `Flight`(`FlightID`), `SeatID` INTEGER NOT NULL REFERENCES `Seat`(`SeatID`), `Is Minor` INTEGER NOT NULL, `BookingID` INTEGER NOT NULL REFERENCES `Booking`(`BookingID`)) STRICT;

PRAGMA FOREIGN_KEYS = ON;

END;
