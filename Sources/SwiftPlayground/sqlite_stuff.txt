BEGIN TRANSACTION;

PRAGMA FOREIGN_KEYS = OFF;

CREATE TABLE IF NOT EXISTS "Customer" (`CustomerID` INTEGER PRIMARY KEY, `Given Name` TEXT NOT NULL CHECK(LENGTH("Given Name") > 0), `Family Name` TEXT NOT NULL CHECK(LENGTH("Family Name") > 0), `Home Phone` TEXT NOT NULL CHECK(LENGTH("Home Phone") > 0), `Work Phone` TEXT, `Mobile Phone` TEXT, `Adress` TEXT NOT NULL, `Region` TEXT NOT NULL, `City` TEXT NOT NULL, `Country` TEXT NOT NULL, `Email` TEXT NOT NULL) STRICT;

CREATE TABLE `Location` (`AirportCode` INTEGER PRIMARY KEY NOT NULL, `Airport Name` TEXT NOT NULL, `Airport Adress` TEXT NOT NULL) STRICT;

CREATE TABLE `Plane` (`PlaneID` INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL, `Model` TEXT NOT NULL, `Min Runway Length` INTEGER NOT NULL, `Max Weight` INTEGER NOT NULL, `Capacity` INTEGER NOT NULL) STRICT;

CREATE TABLE IF NOT EXISTS "Runway" (`RunwayID` INTEGER PRIMARY KEY NOT NULL, `Length` INTEGER NOT NULL, `AirportCode` INTEGER NOT NULL REFERENCES `Location`(`AirportCode`), `Type` TEXT NOT NULL) STRICT;

CREATE TABLE `Seat` (`SeatID` INTEGER PRIMARY KEY AUTOINCREMENT UNIQUE NOT NULL, `PlaneID` INTEGER NOT NULL REFERENCES `Plane`(`PlaneID`), `Seat Class` TEXT NOT NULL, `Emergency` INTEGER NOT NULL) STRICT;

CREATE TABLE `Flight` (`FlightID` INTEGER PRIMARY KEY AUTOINCREMENT UNIQUE NOT NULL, `Departure Time` TEXT NOT NULL DEFAULT (00/00/0000), `PlaneID` INTEGER NOT NULL REFERENCES `Plane`(`PlaneID`), `Departure Location` TEXT NOT NULL REFERENCES `Location`(`AirportCode`), `Destination Location` TEXT NOT NULL REFERENCES `Location`(`AirportCode`)) STRICT;

CREATE TABLE `Staff` (`StaffID` INTEGER PRIMARY KEY AUTOINCREMENT UNIQUE NOT NULL, `Given Name` TEXT NOT NULL, `Family Name` TEXT NOT NULL, `Job Title` TEXT NOT NULL, `Work Email` TEXT NOT NULL, `Work Phone` TEXT NOT NULL, `Mobile Phone` TEXT, `AirportCode` TEXT NOT NULL REFERENCES `Location`(`AirportCode`)) STRICT;

CREATE TABLE `Booking` (`BookingID` INTEGER PRIMARY KEY AUTOINCREMENT NOT NULL, `CustomerID` INTEGER NOT NULL REFERENCES `Customer`(`CustomerID`), `StaffID` INTEGER NOT NULL REFERENCES `Staff`(`StaffID`), `Passenger Count` INTEGER NOT NULL, `FlightID` INTEGER NOT NULL REFERENCES `Flight`(`FlightID`)) STRICT;

CREATE TABLE `Passenger` (`FlightID` INTEGER NOT NULL REFERENCES `Flight`(`FlightID`), `SeatID` INTEGER NOT NULL REFERENCES `Seat`(`SeatID`), `Is Minor` INTEGER NOT NULL, `BookingID` INTEGER NOT NULL REFERENCES `Booking`(`BookingID`)) STRICT;

PRAGMA FOREIGN_KEYS = ON;

END;
